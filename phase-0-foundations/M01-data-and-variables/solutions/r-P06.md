wide <- read.csv("../data/M01_crp_wide.csv")  # خواندن فایل و ساخت جدول

                varying = c("crp_d0", "crp_d3", "crp_d7"),  # نگه‌داشتن مقدار با یک نام
                v.names = "crp", timevar = "day", times = c(0, 3, 7),
                idvar = "id")  # نگه‌داشتن مقدار با یک نام
nrow(long)
